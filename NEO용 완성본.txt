//핀 정의
//왼쪽모터 5V -> 9V 전환 필요 : 흰색 주황색 갈색 선
//오른쪽 모터 5V -> 9V전환필요 : 주황색, 보라색, 회색 선
#define motor_Enable_left   9  // 왼쪽 모터 속도 (PWM 핀)
#define motor_Enable_right  10  // 오른쪽 모터 속도 (PWM 핀)
#define motor_left_f        4  // 왼쪽 모터 정방향
#define motor_left_b        2  // 왼쪽 모터 역방향
#define motor_right_f       7 // 오른쪽 모터 정방향
#define motor_right_b       11  // 오른쪽 모터 역방향

//적외선 센서 핀 정의
int sensor_left = 8;
int sensor_right = 6;

// 초음파 센서 핀 정의
#define trigPin 12
#define echoPin 13

// ==========================================
// [수정된 부분] 고속 주행을 위한 PWM 세팅
// ==========================================
// 주의: 왼쪽 모터 함수에 +10 보정이 있으므로, 
// STRAIGHT_FORWARD를 245 이상으로 올리면 오버플로우(속도 0됨) 발생 위험 있음.

#define STRAIGHT_FORWARD 200 // 직진 속도 (기존 180 -> 210 상향)
#define STRAIGHT_SPEED 230   // 회전 시 바깥쪽 바퀴 속도 (기존 195 -> 230 상향: 확실하게 밀어넣기)
#define REVERSE_SPEED 200    // 회전 시 안쪽 바퀴 역회전 속도 (기존 160 -> 180 상향: 빠른 제동 및 회전)

//모터 제어 함수

//정지
void stopMotors() {
    digitalWrite(motor_left_f, LOW);
    digitalWrite(motor_left_b, LOW);
    digitalWrite(motor_right_f, LOW);
    digitalWrite(motor_right_b, LOW);
    analogWrite(motor_Enable_left, 0);
    analogWrite(motor_Enable_right, 0);
}

//전진
void moveForward(int speed) {
    digitalWrite(motor_left_f, HIGH); 
    digitalWrite(motor_left_b, LOW);
    digitalWrite(motor_right_f, HIGH);
    digitalWrite(motor_right_b, LOW);
    // [주의] speed가 245를 넘으면 여기서 255를 초과하여 에러(느려짐) 발생 가능
    analogWrite(motor_Enable_left, speed); 
    analogWrite(motor_Enable_right, speed);
}

//좌회전
void turnLeftHard(int speedS, int speedR) {
    // 왼쪽 역방향, 오른쪽 정방향
    digitalWrite(motor_left_f, LOW);
    digitalWrite(motor_left_b, HIGH);
    digitalWrite(motor_right_f, HIGH);
    digitalWrite(motor_right_b, LOW);
    analogWrite(motor_Enable_left, speedR); //정방향 역방향 속도 차이 고려
    analogWrite(motor_Enable_right, speedS);
}

//우회전
void turnRightHard(int speedS, int speedR) {
    // 왼쪽 정방향, 오른쪽 역방향
    digitalWrite(motor_left_f, HIGH);
    digitalWrite(motor_left_b, LOW);
    digitalWrite(motor_right_f, LOW);
    digitalWrite(motor_right_b, HIGH);
    analogWrite(motor_Enable_left, speedS);
    analogWrite(motor_Enable_right, speedR); //정방향 역방향 속도 차이 고려
}

//초음파센서
long StartSensor() {
    digitalWrite(trigPin, LOW);
    delayMicroseconds(2);
    digitalWrite(trigPin, HIGH);
    delayMicroseconds(10);
    digitalWrite(trigPin, LOW);
    long duration = pulseIn(echoPin, HIGH);
    long distance = duration / 29 / 2; //초음파센서 단위환산 : us -> cm
    return distance;
}

void setup()
{
    pinMode(sensor_left, INPUT);
    pinMode(sensor_right, INPUT);

    pinMode(motor_left_f, OUTPUT);
    pinMode(motor_left_b, OUTPUT);
    pinMode(motor_right_f, OUTPUT);
    pinMode(motor_right_b, OUTPUT);
    
    pinMode(trigPin, OUTPUT);
    pinMode(echoPin, INPUT);
    stopMotors(); //정지된 상태로 시작
    while (true) {
        long distance = StartSensor();
        // 거리가 10cm 이내, 0cm 보다 크면, 반복문 벗어나고 출발
        if (distance < 10 && distance > 0) {
            delay(1000);
            break;
        }
        delay(100); // 0.1초마다 거리 확인
    }
    
    // [수정] 출발 시 확실한 구동을 위해 255(최대)로 짧게 밀어줌
    moveForward(255); 
    delay(250);
}

void loop()
{
    
    // 센서 값 읽기 (흰색=0 / 검은색=1)
    int s_l = digitalRead(sensor_left);
    int s_r = digitalRead(sensor_right);

    // 1. (0, 0) : 양쪽 모두 흰색 -> 직진 (라인을 가운데에 두고 잘 가는 중)
    if (s_l == 0 && s_r == 0)
    {
        moveForward(STRAIGHT_FORWARD);
    }
    
    // 2. (0, 1) : 오른쪽 센서가 검은색 인식 -> 우회전 (차가 왼쪽으로 쏠림)
    else if(s_l == 0 && s_r == 1)
    {
        turnRightHard(STRAIGHT_SPEED, REVERSE_SPEED);
    }

    // 3. (1, 0) : 왼쪽 센서가 검은색 인식 -> 좌회전 (차가 오른쪽으로 쏠림)
    else if(s_l == 1 && s_r == 0)
    {
        // 왼쪽으로 꺾어야 함 (왼쪽 바퀴 정지)
        turnLeftHard(STRAIGHT_SPEED+25, REVERSE_SPEED+55);
    }

    // 4. (1, 1) : 양쪽 모두 검은색 -> 정지 (교차선 또는 오류)
    else
    {
        stopMotors();
    }
}
